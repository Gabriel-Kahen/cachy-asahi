
FRESH607_CLOCK_CONTEXT = None

def fresh607_clock_binding(c):
    """Source-owned current workflow context, original nonce and owned workspace."""
    context=FRESH607_CLOCK_CONTEXT
    need(type(context)is dict and set(context)=={'schema_version','purpose','source','run_id','run_attempt'} and
         type(context['schema_version'])is int and context['schema_version']==1 and
         context['purpose']=='SOURCE_OWNED_GENERIC16K_REALTIME_INIT_V1' and
         type(context['source'])is str and re.fullmatch('[0-9a-f]{40}',context['source']) and
         type(context['run_id'])is str and re.fullmatch('[1-9][0-9]{0,19}',context['run_id']) and
         type(context['run_attempt'])is str and re.fullmatch('[1-9][0-9]{0,5}',context['run_attempt']),
         'closed current fresh607 realtime context')
    value=c.get('fresh607_realtime')
    need(type(value)is dict and set(value)==set(context)|{'nonce','workspace','workspace_identity','created_realtime_ns'} and
         all(value[k]==v for k,v in context.items()) and value['nonce']==c['nonce'] and
         type(value['nonce'])is str and re.fullmatch('[0-9a-f]{32}',value['nonce']) and
         type(value['workspace'])is str and value['workspace'].startswith('/tmp/') and
         len(value['workspace'].encode())<=4096 and '..'not in Path(value['workspace']).parts and
         str(Path(value['workspace']))==value['workspace'] and
         type(value['workspace_identity'])is list and len(value['workspace_identity'])==4 and
         all(type(n)is int and n>=0 for n in value['workspace_identity']) and
         type(value['created_realtime_ns'])is int and 1_000_000_000_000_000_000<=value['created_realtime_ns']<2**63,
         'original nonce/current run/source/owned workspace realtime binding')
    return value

def fresh607_clock_contract(c,output):
    import stat
    info=output.lstat()
    need(not output.is_symlink() and output.parent==Path('/tmp') and
         stat.S_ISDIR(info.st_mode) and stat.S_IMODE(info.st_mode)==0o700 and
         info.st_uid==os.getuid() and info.st_gid==os.getgid(),
         'same fresh owned controller workspace before clock capture')
    c['fresh607_realtime']=dict(FRESH607_CLOCK_CONTEXT,nonce=c['nonce'],workspace=str(output),
        workspace_identity=[info.st_dev,info.st_ino,info.st_uid,info.st_gid],created_realtime_ns=time.time_ns())
    fresh607_clock_binding(c)
    need((output.lstat().st_dev,output.lstat().st_ino)==(info.st_dev,info.st_ino),'clock capture workspace identity drift')

def fresh607_clock_argument(c):
    value=fresh607_clock_binding(c);now=time.time_ns()
    need(value['created_realtime_ns']<=now<=value['created_realtime_ns']+9_600_000_000_000,
         'launch current host realtime inside original VM owner lifetime')
    binding=hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True).encode('ascii')).hexdigest()
    return ' caa_fresh607_realtime_ns='+str(now)+' caa_fresh607_clock_sha256='+binding
