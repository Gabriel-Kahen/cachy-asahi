function(require_imported_location target expected)
    if(NOT TARGET ${target})
        message(FATAL_ERROR "Required normal imported target is missing: ${target}")
    endif()
    get_target_property(_imported ${target} IMPORTED)
    get_target_property(_configs ${target} IMPORTED_CONFIGURATIONS)
    get_target_property(_generic ${target} IMPORTED_LOCATION)
    set(_locations)
    if(_generic)
        list(APPEND _locations "${_generic}")
    endif()
    if(_configs)
        foreach(_configuration IN LISTS _configs)
            string(TOUPPER "${_configuration}" _configuration)
            get_target_property(_location ${target} IMPORTED_LOCATION_${_configuration})
            list(APPEND _locations "${_location}")
        endforeach()
    endif()
    if(NOT _imported OR NOT _locations)
        message(FATAL_ERROR "Required imported target has no location: ${target}")
    endif()
    foreach(_location IN LISTS _locations)
        if(NOT _location STREQUAL expected OR NOT EXISTS "${_location}")
            message(FATAL_ERROR "Refusing missing/foreign imported target: ${target}")
        endif()
        file(REAL_PATH "${_location}" _real)
        if(NOT _real STREQUAL expected)
            message(FATAL_ERROR "Refusing aliased imported target: ${target}")
        endif()
    endforeach()
endfunction()

foreach(_module IN LISTS _dependencies _own_modules)
    set(_prefix "${QT612_DEPENDENCY_PREFIX}")
    if(_module IN_LIST _own_modules)
        set(_prefix "${QT612_BUILD_TREE}")
    endif()
    if(NOT Qt6${_module}_DIR STREQUAL "${_prefix}/lib/cmake/Qt6${_module}")
        message(FATAL_ERROR "Refusing foreign Qt module discovery: ${_module}")
    endif()
    get_target_property(_version Qt6::${_module} _qt_package_version)
    if(NOT _version STREQUAL "6.12.0")
        message(FATAL_ERROR "Exact Qt6.12 module metadata is required")
    endif()
    string(REGEX REPLACE "Private$" "" _library "${_module}")
    require_imported_location(Qt6::${_module} "${_prefix}/lib/libQt6${_library}.so.6.12.0")
endforeach()
