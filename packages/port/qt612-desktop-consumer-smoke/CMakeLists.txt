function(require_location target expected)
    if(NOT TARGET ${target})
        message(FATAL_ERROR "Missing installed target: ${target}")
    endif()
    get_target_property(imported ${target} IMPORTED)
    get_target_property(configs ${target} IMPORTED_CONFIGURATIONS)
    get_target_property(location ${target} IMPORTED_LOCATION)
    set(locations)
    if(location)
        list(APPEND locations "${location}")
    endif()
    if(configs)
        foreach(config IN LISTS configs)
            string(TOUPPER "${config}" config)
            get_target_property(location ${target} IMPORTED_LOCATION_${config})
            list(APPEND locations "${location}")
        endforeach()
    endif()
    if(NOT imported OR NOT locations)
        message(FATAL_ERROR "No imported installed location: ${target}")
    endif()
    foreach(location IN LISTS locations)
        file(REAL_PATH "${location}" real)
        if(NOT location STREQUAL expected OR NOT real STREQUAL expected OR NOT EXISTS "${location}")
            message(FATAL_ERROR "Foreign or absent installed provider: ${target}: ${location}")
        endif()
    endforeach()
endfunction()

foreach(module IN LISTS modules)
    get_target_property(version Qt6::${module} _qt_package_version)
    if(NOT version STREQUAL "6.12.0" OR NOT Qt6${module}_DIR STREQUAL "/usr/lib/cmake/Qt6${module}")
        message(FATAL_ERROR "Exact installed Qt6.12 provider required: ${module}")
    endif()
    require_location(Qt6::${module} /usr/lib/libQt6${module}.so.6.12.0)
endforeach()

# The two upstream macros use the same C output basename. Generate in separate
# directories and link the shared protocol definition once, from the client.
add_subdirectory(protocol-client)
add_subdirectory(protocol-server)
add_executable(qt612-desktop-protocol protocol.cpp)
target_compile_features(qt612-desktop-protocol PRIVATE cxx_std_17)
target_include_directories(qt612-desktop-protocol PRIVATE
    ${CMAKE_CURRENT_BINARY_DIR}/protocol-client ${CMAKE_CURRENT_BINARY_DIR}/protocol-server)
target_link_libraries(qt612-desktop-protocol PRIVATE qt612-protocol-client qt612-protocol-server
    Qt6::Core Qt6::WaylandClient Qt6::WaylandCompositor Wayland::Client Wayland::Server)
add_dependencies(qt612-desktop-protocol qt612-protocol-client qt612-protocol-server)
