cmake_minimum_required(VERSION 3.25)
project(qt612_declarative_smoke LANGUAGES CXX)
if(NOT IS_ABSOLUTE "${QT612_BUILD_TREE}" OR NOT IS_DIRECTORY "${QT612_BUILD_TREE}/lib/cmake/Qt6Qml")
    message(FATAL_ERROR "Explicit own Qt Declarative build tree required")
endif()
file(REAL_PATH "${QT612_BUILD_TREE}" actual)
if(NOT actual STREQUAL QT612_BUILD_TREE)
    message(FATAL_ERROR "Canonical build tree required")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
foreach(module IN ITEMS Core Gui Network Widgets Test OpenGL Sql)
    set(Qt6${module}_DIR "/usr/lib/cmake/Qt6${module}" CACHE PATH "" FORCE)
endforeach()
foreach(module IN ITEMS Qml QmlPrivate QmlIntegration Quick QuickPrivate QuickControls2 QuickTemplates2 QuickLayouts QuickDialogs2 QuickTest)
    set(Qt6${module}_DIR "${QT612_BUILD_TREE}/lib/cmake/Qt6${module}" CACHE PATH "" FORCE)
    find_package(Qt6${module} 6.12.0 EXACT CONFIG REQUIRED PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6${module}" NO_DEFAULT_PATH)
endforeach()
find_package(Qt6 6.12.0 EXACT CONFIG REQUIRED COMPONENTS Core Gui Network Widgets Test OpenGL Sql PATHS /usr/lib/cmake/Qt6 NO_DEFAULT_PATH)
if(NOT COMMAND qt6_add_qml_module)
    message(FATAL_ERROR "Installed/exported QML CMake API is required")
endif()
add_executable(qt612-declarative-smoke smoke.cpp)
target_compile_features(qt612-declarative-smoke PRIVATE cxx_std_17)
target_compile_definitions(qt612-declarative-smoke PRIVATE QT612_OWN_QML="${QT612_BUILD_TREE}/lib/qt6/qml")
target_link_libraries(qt612-declarative-smoke PRIVATE Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::QuickTemplates2 Qt6::QuickLayouts Qt6::QuickDialogs2 Qt6::QuickTest Qt6::Gui Qt6::Widgets)
# Exercise normal consumer QML code generation and qmltyperegistrar, not only headers.
qt6_add_qml_module(qt612-declarative-smoke URI Reference.Consumer VERSION 1.0 QML_FILES Consumer.qml)
