cmake_minimum_required(VERSION 3.25)
project(qt612_reference_smoke LANGUAGES CXX)
if(NOT IS_DIRECTORY "${QT612_BUILD_TREE}/lib/cmake/Qt6")
    message(FATAL_ERROR "An explicit own-built Qt6 tree is required")
endif()
set(Qt6_DIR "${QT612_BUILD_TREE}/lib/cmake/Qt6" CACHE PATH "" FORCE)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Network Widgets)
if(NOT Qt6_DIR STREQUAL "${QT612_BUILD_TREE}/lib/cmake/Qt6")
    message(FATAL_ERROR "Refusing a different Qt installation")
endif()
add_executable(qt612-smoke smoke.cpp)
target_link_libraries(qt612-smoke PRIVATE Qt6::Core Qt6::Network Qt6::Widgets)
target_compile_features(qt612-smoke PRIVATE cxx_std_17)
