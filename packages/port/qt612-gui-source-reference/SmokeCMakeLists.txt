cmake_minimum_required(VERSION 3.25)
project(qt612_gui_reference_smoke LANGUAGES CXX)
if(NOT IS_ABSOLUTE "${QT612_BUILD_TREE}" OR
   NOT IS_DIRECTORY "${QT612_BUILD_TREE}/lib/cmake/Qt6")
    message(FATAL_ERROR "An explicit absolute own-built Qt6 tree is required")
endif()
file(REAL_PATH "${QT612_BUILD_TREE}" _qt612_tree)
if(NOT QT612_BUILD_TREE STREQUAL _qt612_tree)
    message(FATAL_ERROR "The own-built Qt6 tree must use its canonical path")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(_qt612_modules Core Gui Network Widgets OpenGL OpenGLWidgets PrintSupport WaylandClient)
set(_qt612_private_modules EglFSDeviceIntegrationPrivate EglFsKmsGbmSupportPrivate
                          EglFsKmsSupportPrivate WlShellIntegrationPrivate)
set(_qt612_all_modules ${_qt612_modules} ${_qt612_private_modules})
foreach(_module IN LISTS _qt612_all_modules)
    set(_config "${QT612_BUILD_TREE}/lib/cmake/Qt6${_module}")
    if(NOT EXISTS "${_config}/Qt6${_module}Config.cmake")
        message(FATAL_ERROR "Missing own-built Qt6${_module} discovery metadata")
    endif()
    set(Qt6${_module}_DIR "${_config}" CACHE PATH "" FORCE)
endforeach()
set(Qt6_DIR "${QT612_BUILD_TREE}/lib/cmake/Qt6" CACHE PATH "" FORCE)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Network Widgets OpenGL OpenGLWidgets PrintSupport WaylandClient
             CONFIG PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6" NO_DEFAULT_PATH)
find_package(Qt6EglFSDeviceIntegrationPrivate 6.12.0 EXACT REQUIRED CONFIG
             PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6EglFSDeviceIntegrationPrivate" NO_DEFAULT_PATH)
find_package(Qt6EglFsKmsGbmSupportPrivate 6.12.0 EXACT REQUIRED CONFIG
             PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6EglFsKmsGbmSupportPrivate" NO_DEFAULT_PATH)
find_package(Qt6EglFsKmsSupportPrivate 6.12.0 EXACT REQUIRED CONFIG
             PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6EglFsKmsSupportPrivate" NO_DEFAULT_PATH)
find_package(Qt6WlShellIntegrationPrivate 6.12.0 EXACT REQUIRED CONFIG
             PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6WlShellIntegrationPrivate" NO_DEFAULT_PATH)
if(NOT Qt6_DIR STREQUAL "${QT612_BUILD_TREE}/lib/cmake/Qt6")
    message(FATAL_ERROR "Refusing a different Qt installation")
endif()
foreach(_module IN LISTS _qt612_all_modules)
    if(NOT Qt6${_module}_DIR STREQUAL "${QT612_BUILD_TREE}/lib/cmake/Qt6${_module}" OR
       NOT TARGET Qt6::${_module})
        message(FATAL_ERROR "Refusing foreign Qt6${_module} discovery")
    endif()
    get_target_property(_imported Qt6::${_module} IMPORTED)
    get_target_property(_version Qt6::${_module} _qt_package_version)
    get_target_property(_configs Qt6::${_module} IMPORTED_CONFIGURATIONS)
    if(NOT _imported OR NOT _version STREQUAL "6.12.0" OR NOT _configs)
        message(FATAL_ERROR "An imported own-built Qt6.12 module is required")
    endif()
    # QtModuleHelpers strips Private from INTERNAL_MODULE library names.
    string(REGEX REPLACE "Private$" "" _library_module "${_module}")
    set(_expected "${QT612_BUILD_TREE}/lib/libQt6${_library_module}.so.6.12.0")
    foreach(_configuration IN LISTS _configs)
        string(TOUPPER "${_configuration}" _configuration)
        get_target_property(_location Qt6::${_module} IMPORTED_LOCATION_${_configuration})
        if(NOT _location STREQUAL _expected OR NOT EXISTS "${_location}")
            message(FATAL_ERROR "Refusing a missing or foreign Qt6${_module} imported library")
        endif()
        file(REAL_PATH "${_location}" _real_location)
        if(NOT _real_location STREQUAL _expected)
            message(FATAL_ERROR "Refusing an aliased Qt6${_module} imported library")
        endif()
    endforeach()
    get_target_property(_generic_location Qt6::${_module} IMPORTED_LOCATION)
    if(_generic_location AND NOT _generic_location STREQUAL _expected)
        message(FATAL_ERROR "Refusing a foreign generic Qt6${_module} imported library")
    endif()
endforeach()
add_executable(qt612-smoke smoke.cpp)
target_link_libraries(qt612-smoke PRIVATE Qt6::Core Qt6::Network Qt6::Widgets
                      Qt6::OpenGL Qt6::OpenGLWidgets Qt6::PrintSupport Qt6::WaylandClient)
target_compile_features(qt612-smoke PRIVATE cxx_std_17)
