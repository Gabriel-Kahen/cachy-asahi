cmake_minimum_required(VERSION 3.25)
project(qt612_svg_reference_smoke LANGUAGES CXX)
foreach(_root IN ITEMS QT612_BUILD_TREE QT612_DEPENDENCY_PREFIX)
    if(NOT IS_ABSOLUTE "${${_root}}" OR NOT IS_DIRECTORY "${${_root}}/lib/cmake")
        message(FATAL_ERROR "An explicit absolute ${_root} is required")
    endif()
    file(REAL_PATH "${${_root}}" _canonical)
    if(NOT "${${_root}}" STREQUAL "${_canonical}")
        message(FATAL_ERROR "${_root} must use its canonical path")
    endif()
endforeach()
if(QT612_BUILD_TREE STREQUAL QT612_DEPENDENCY_PREFIX)
    message(FATAL_ERROR "Own SVG build tree and base dependency prefix must differ")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(_qt612_base_modules Core Gui Widgets)
set(_qt612_svg_modules Svg SvgWidgets)
foreach(_module IN LISTS _qt612_base_modules _qt612_svg_modules)
    if(_module IN_LIST _qt612_svg_modules)
        set(_root "${QT612_BUILD_TREE}")
    else()
        set(_root "${QT612_DEPENDENCY_PREFIX}")
    endif()
    set(_config "${_root}/lib/cmake/Qt6${_module}")
    if(NOT EXISTS "${_config}/Qt6${_module}Config.cmake")
        message(FATAL_ERROR "Missing explicit Qt6${_module} discovery metadata")
    endif()
    set(Qt6${_module}_DIR "${_config}" CACHE PATH "" FORCE)
endforeach()
set(Qt6_DIR "${QT612_DEPENDENCY_PREFIX}/lib/cmake/Qt6" CACHE PATH "" FORCE)
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Gui Widgets CONFIG
             PATHS "${QT612_DEPENDENCY_PREFIX}/lib/cmake/Qt6" NO_DEFAULT_PATH)
find_package(Qt6Svg 6.12.0 EXACT REQUIRED CONFIG
             PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6Svg" NO_DEFAULT_PATH)
find_package(Qt6SvgWidgets 6.12.0 EXACT REQUIRED CONFIG
             PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6SvgWidgets" NO_DEFAULT_PATH)
if(NOT Qt6_DIR STREQUAL "${QT612_DEPENDENCY_PREFIX}/lib/cmake/Qt6")
    message(FATAL_ERROR "Refusing a foreign Qt umbrella package")
endif()
function(qt612_require_imported _module _root)
    if(NOT Qt6${_module}_DIR STREQUAL "${_root}/lib/cmake/Qt6${_module}" OR
       NOT TARGET Qt6::${_module})
        message(FATAL_ERROR "Refusing foreign Qt6${_module} discovery")
    endif()
    get_target_property(_imported Qt6::${_module} IMPORTED)
    get_target_property(_version Qt6::${_module} _qt_package_version)
    get_target_property(_configs Qt6::${_module} IMPORTED_CONFIGURATIONS)
    if(NOT _imported OR NOT _version STREQUAL "6.12.0" OR NOT _configs)
        message(FATAL_ERROR "An imported exact Qt6.12 module is required")
    endif()
    set(_expected "${_root}/lib/libQt6${_module}.so.6.12.0")
    foreach(_configuration IN LISTS _configs)
        string(TOUPPER "${_configuration}" _configuration)
        get_target_property(_location Qt6::${_module} IMPORTED_LOCATION_${_configuration})
        if(NOT _location STREQUAL _expected OR NOT EXISTS "${_location}")
            message(FATAL_ERROR "Refusing a missing or foreign Qt6${_module} imported library")
        endif()
        file(REAL_PATH "${_location}" _real_location)
        if(NOT _real_location STREQUAL _expected)
            message(FATAL_ERROR "Refusing an aliased Qt6${_module} imported library")
        endif()
    endforeach()
    get_target_property(_generic_location Qt6::${_module} IMPORTED_LOCATION)
    if(_generic_location AND NOT _generic_location STREQUAL _expected)
        message(FATAL_ERROR "Refusing foreign generic Qt6${_module} imported library")
    endif()
endfunction()
foreach(_module IN LISTS _qt612_base_modules)
    qt612_require_imported("${_module}" "${QT612_DEPENDENCY_PREFIX}")
endforeach()
foreach(_module IN LISTS _qt612_svg_modules)
    qt612_require_imported("${_module}" "${QT612_BUILD_TREE}")
endforeach()
add_executable(qt612-smoke smoke.cpp)
target_link_libraries(qt612-smoke PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Svg Qt6::SvgWidgets)
target_compile_features(qt612-smoke PRIVATE cxx_std_17)
target_compile_definitions(qt612-smoke PRIVATE
    QT612_EXPECTED_PLUGIN_ROOT=\"${QT612_BUILD_TREE}/lib/qt6/plugins\")
