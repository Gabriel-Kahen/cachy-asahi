cmake_minimum_required(VERSION 3.25)
project(qt612_shadertools_finite_smoke LANGUAGES CXX)
if(NOT IS_ABSOLUTE "${QT612_BUILD_TREE}" OR NOT IS_DIRECTORY "${QT612_BUILD_TREE}/lib/cmake/Qt6ShaderTools")
    message(FATAL_ERROR "Explicit own ShaderTools build tree required")
endif()
file(REAL_PATH "${QT612_BUILD_TREE}" actual)
if(NOT actual STREQUAL QT612_BUILD_TREE)
    message(FATAL_ERROR "Canonical own build tree required")
endif()
if(QT_NO_PACKAGE_VERSION_CHECK)
    message(FATAL_ERROR "Normal Qt version checks required")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
foreach(module IN ITEMS Core Gui)
    set(Qt6${module}_DIR "/usr/lib/cmake/Qt6${module}" CACHE PATH "" FORCE)
    find_package(Qt6${module} 6.12.0 EXACT CONFIG REQUIRED PATHS "/usr/lib/cmake/Qt6${module}" NO_DEFAULT_PATH)
endforeach()
foreach(module IN ITEMS ShaderTools ShaderToolsPrivate ShaderToolsTools)
    set(Qt6${module}_DIR "${QT612_BUILD_TREE}/lib/cmake/Qt6${module}" CACHE PATH "" FORCE)
    find_package(Qt6${module} 6.12.0 EXACT CONFIG REQUIRED PATHS "${QT612_BUILD_TREE}/lib/cmake/Qt6${module}" NO_DEFAULT_PATH)
endforeach()
if(NOT TARGET Qt6::qsb OR NOT TARGET Qt6::ShaderTools OR NOT TARGET Qt6::ShaderToolsPrivate OR NOT COMMAND qt6_add_shaders)
    message(FATAL_ERROR "Own qsb, private/public targets and exported shader macro are mandatory")
endif()
get_target_property(qsb Qt6::qsb IMPORTED_LOCATION)
if(NOT qsb)
    get_target_property(qsb Qt6::qsb IMPORTED_LOCATION_RELEASE)
endif()
if(NOT qsb STREQUAL "${QT612_BUILD_TREE}/lib/qt6/bin/qsb" OR NOT EXISTS "${qsb}")
    message(FATAL_ERROR "No fallback to a foreign host qsb")
endif()
add_executable(qt612-shader-api smoke.cpp)
target_compile_features(qt612-shader-api PRIVATE cxx_std_17)
target_link_libraries(qt612-shader-api PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate Qt6::ShaderTools Qt6::ShaderToolsPrivate)
qt6_add_shaders(qt612-shader-api finite_vertex BATCHABLE PREFIX /finite
    FILES smoke.vert)
qt6_add_shaders(qt612-shader-api finite_fragment PREFIX /finite
    FILES smoke.frag)
